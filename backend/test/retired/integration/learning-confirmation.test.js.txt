async function fixture(confirmed = false) {
  const id = randomUUID(), versionId = randomUUID(), reviewId = randomUUID();
  const classification = { typeId: 'procurement', tags: [], confirmed: true, perspectiveConfirmed: true, versionId };
  const snapshot = { typeId: 'procurement', classification, background: {}, rulePacks: [], skillPacks: [] };
  await db('la_contracts').insert({ id, number: 'CONFIRM-' + id, title: '确认门槛回归样本', classification: JSON.stringify(classification) });
  await db('la_versions').insert({ id: versionId, contract_id: id, version_no: 1, kind: 'contract', filename: 'fixture.docx', storage_key: 'objects/fixture', sha256: 'a'.repeat(64), size: 10, media_type: 'application/docx', source_action: 'fixture', parsed: JSON.stringify({ blocks: [{ id: 'b1', text: '验收后付款。' }] }) });
  await db('la_reviews').insert({ id: reviewId, contract_id: id, version_id: versionId, status: 'completed', coverage: '{"complete":true}', snapshot: JSON.stringify(snapshot) });
  await db('la_contracts').where({ id }).update({ current_version_id: versionId, current_review_id: reviewId, legal_status: confirmed ? 'finalized' : 'pending_review' });
  let decision;
  if (confirmed) [decision] = await db('la_decisions').insert({ id: randomUUID(), review_id: reviewId, version_id: versionId, decided_by: user.id, outcome: 'agree' }).returning('*');
  return { contract: await db('la_contracts').where({ id }).first(), run: await db('la_reviews').where({ id: reviewId }).first(), versionId, snapshot, decision };
}
async function cachedTask(f, decision = f.decision) {
  const id = randomUUID();
  const source = { documents: [{ id: f.versionId, versionNo: 1, blocks: [{ id: 'b1', text: '验收后付款。' }] }], decision: { id: decision.id, outcome: 'agree' }, findings: [], comparison: 'no_previous_version', restrictedTerms: [] };
  await db('la_learning_tasks').insert({ id, decision_id: decision.id, contract_id: f.contract.id, review_id: f.run.id, version_id: f.versionId, created_by: user.id, snapshot: JSON.stringify(f.snapshot), source: JSON.stringify(source) });
  return id;
}

  const f = await fixture(), id = randomUUID();

  const unconfirmed = await fixture(), other = await fixture(true), id = await cachedTask(unconfirmed, other.decision); let calls = 0;

  const retry = await call('post', `/learning-tasks/${id}/retry`, { revision: task.revision }).expect(409); assert.equal(retry.body.code, 'LEARNING_CONFIRMATION_REQUIRED');

test('authorized withdrawal preserves the decision and evidence, pauses its skills, and requires a new legal review', async () => {
  const { withdrawForReconfirmation } = require('../../pilot/decision-reconfirmation');
  const { createSkillDraft } = require('../../pilot/skills');
  const f = await fixture(true), id = await cachedTask(f), originalDecision = await db('la_decisions').where({ id: f.decision.id }).first(), originalTask = await db('la_learning_tasks').where({ id }).first();
  const pack = await db.transaction(trx => createSkillDraft(trx, { name: '撤回来源样例', scope: { level: 'type', typeIds: ['procurement'] }, skills: [{ id: 'withdraw-check', title: '验收检查', method: '核对验收条件', conditions: [], exclusions: [], examples: [], sourceIds: [id] }] }, user.id));
  await db('la_skill_versions').where({ id: pack.id }).update({ state: 'published' });
  const result = await withdrawForReconfirmation(db, user, f.decision.id, '用户明确说明原结论未经法务实际确认，要求重新复核');
  assert.equal(result.withdrawn, true); assert.equal(result.pausedTasks, 1); assert.equal(result.retiredVersions.length, 1);
  const current = await db('la_contracts').where({ id: f.contract.id }).first(); assert.equal(current.legal_status, 'pending_review'); assert.equal(current.current_review_id, null); assert.equal(current.current_version_id, f.versionId);
  assert.deepEqual(await db('la_decisions').where({ id: f.decision.id }).first(), originalDecision);
  const paused = await db('la_learning_tasks').where({ id }).first(); assert.deepEqual(paused.source, originalTask.source); assert.deepEqual(paused.proposals, originalTask.proposals); assert.equal(paused.status, 'failed');
  assert.equal((await db('la_skill_versions').where({ id: pack.id }).first()).state, 'retired');
  assert.equal((await call('get', `/reviews/${f.run.id}`).expect(200)).body.decision.withdrawal.reason, '用户明确说明原结论未经法务实际确认，要求重新复核');
  assert.equal((await call('get', `/reviews/${f.run.id}/final-contract`).expect(409)).body.code, 'DECISION_WITHDRAWN');
  const report = await call('get', `/reviews/${f.run.id}/report?format=html`).expect(200); assert.match(report.text, /最终结论已撤回/); assert.ok(!report.text.includes('<b>同意</b>'));
  await call('post', `/learning-tasks/${id}/retry`, { revision: paused.revision }).expect(409);
  assert.equal((await withdrawForReconfirmation(db, user, f.decision.id, '重复调用')).idempotent, true);
  // Copying the old methods into a fresh draft must not bypass withdrawal.
  const copy = await db.transaction(trx => createSkillDraft(trx, { name: '撤回后复制样例', scope: pack.scope, skills: pack.skills }, user.id));
  assert.equal((await call('post', `/skill-pack-versions/${copy.id}/submit`, { revision: copy.revision }).expect(409)).body.code, 'LEARNING_CONFIRMATION_REQUIRED');
  const reviewId = randomUUID(); await db('la_reviews').insert({ id: reviewId, contract_id: f.contract.id, version_id: f.versionId, status: 'completed', coverage: '{"complete":true}', snapshot: JSON.stringify(f.snapshot) });
  await db('la_contracts').where({ id: f.contract.id }).update({ current_review_id: reviewId });
  const c = await db('la_contracts').where({ id: f.contract.id }).first();
  const confirmed = await call('post', `/reviews/${reviewId}/decision`, { revision: c.revision, outcome: 'agree', reason: '法务重新核对后确认' }).expect(201);
  assert.notEqual(confirmed.body.id, f.decision.id); assert.ok(await db('la_learning_tasks').where({ decision_id: confirmed.body.id }).first());
  assert.equal((await db('la_decisions').whereIn('id', [f.decision.id, confirmed.body.id])).length, 2);
});
