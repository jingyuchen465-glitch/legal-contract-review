  assert.equal((await db('la_reviews')).length, 0); assert.equal(formalCalls.length, 0);
  assert.equal((await db('la_outbox').whereRaw("payload->>'preanalysisId' = ?", [item.preanalysisId]).first()).queue, 'preanalysis');

test('worker restart requeues interrupted attempts, preserves their snapshots, and leaves live workers alone', async () => {
  const entry = await upload(), id = entry.preanalysisId;
  await db('la_outbox').whereRaw("payload->>'preanalysisId' = ?", [id]).update({ sent_at: db.fn.now() });
  await db('la_preanalyses').where({ id }).update({ status: 'running', stage: 'identifying', progress: 50 });
  const snapshot = (await db('la_preanalyses').where({ id }).first()).snapshot;
  const connection = await db.client.acquireConnection();
  try {
    await db.raw('SELECT pg_advisory_lock(hashtext(?))', [`preanalysis:${id}`]).connection(connection);
    assert.equal(await recoverInterruptedPreanalyses(db), 0);
    assert.equal((await db('la_preanalyses').where({ id }).first()).status, 'running');
  } finally { await db.raw('SELECT pg_advisory_unlock(hashtext(?))', [`preanalysis:${id}`]).connection(connection); await db.client.releaseConnection(connection); }
  assert.equal(await recoverInterruptedPreanalyses(db), 1);
  assert.equal(await recoverInterruptedPreanalyses(db), 0);
  const recovered = await db('la_preanalyses').where({ id }).first();
  assert.equal(recovered.status, 'queued'); assert.deepEqual(recovered.snapshot, snapshot);
  assert.equal((await db('la_outbox').whereNull('sent_at').whereRaw("payload->>'preanalysisId' = ?", [id])).length, 1);
  await processPreanalysis(db, id, { gateway }); assert.equal((await get(entry)).preanalysis.status, 'ready');
});
test('Office conversion proceeds independently while AI slots are occupied and HTTP remains responsive', async () => {
  const entries = [await upload(), await upload()]; let count = 0, release;
  const gate = new Promise(resolve => { release = resolve; });
  // Earlier fixtures have no queued work: only these two jobs are dispatched.
  await db('la_outbox').whereNull('sent_at').whereNotIn(db.raw("payload->>'preanalysisId'"), entries.map(x => x.preanalysisId)).update({ sent_at: db.fn.now() });
  await db('la_outbox').whereNull('sent_at').where({ queue: 'review' }).update({ sent_at: db.fn.now() });
  let conversions = 0;
  const converted = new AdmZip();
  converted.addFile('manifest.json', Buffer.from(JSON.stringify({ version: 1, sourceExtension: '.xlsx', notes: [], warnings: [], document: { parserVersion: 'converter-test', format: 'spreadsheet', blocks: [{ id: 'b1', text, kind: 'table_cell', source: 'spreadsheet', page: null }], sheets: [{ name: '采购清单', state: 'visible' }], requiresHumanVerification: true, warnings: [] } })));
  converted.addFile('preview.pdf', Buffer.from('%PDF-1.4\n% fixture'));
  queue = await createQueue(db, { conversionOptions: { convert: async () => { conversions++; return converted.toBuffer(); } }, preanalysisOptions: { gateway: { ...gateway, preAnalyze: async d => { count++; await gate; return gateway.preAnalyze(d); } } }, reviewOptions: { gateway } });
  const wait = async predicate => { const until = Date.now() + 12000; while (!await predicate()) { if (Date.now() > until) throw Error('queue timeout'); await new Promise(resolve => setTimeout(resolve, 40)); } };
  try {
    await wait(() => count >= 2);
    const started = Date.now(); await owner.get('/api/contracts').expect(200); assert.ok(Date.now() - started < 2000);
    for (const entry of entries) assert.equal((await get(entry)).preanalysis.status, 'running');
    const workbook = new AdmZip(); workbook.addFile('xl/workbook.xml', Buffer.from('<workbook/>'));
    const office = (await owner.post('/api/contracts/upload').set('X-CSRF-Token', csrf).field('reviewFlow', 'guided').attach('file', workbook.toBuffer(), '后台转换.xlsx').expect(201)).body;
    entries.push(office);
    await wait(async () => !!(await db('la_versions').where({ id: office.version.id }).first()).parsed);
    const version = await db('la_versions').where({ id: office.version.id }).first();
    assert.equal(version.conversion.status, 'completed'); assert.equal(conversions, 1);
    assert.equal((await get(office)).preanalysis.status, 'queued'); assert.equal(count, 2);
  } finally { release(); }
  await wait(async () => (await Promise.all(entries.map(get))).every(c => c.preanalysis.status === 'ready'));
  for (const entry of entries) assert.equal((await get(entry)).current_review_id, null);
});
