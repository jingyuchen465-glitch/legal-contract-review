const { test } = require('node:test');
const assert = require('node:assert/strict');
const { setTimeout: delay } = require('node:timers/promises');
const { config } = require('../pilot/config');
const { createGateway } = require('../pilot/model');

const document = { blocks: [{ id: 'b1', text: '甲方验收后付款。' }] };
const rules = Array.from({ length: 26 }, (_, i) => ({ id: `r${i}`, title: `规则${i}` }));
function response(input) {
  return Response.json({ choices: [{ message: { content: JSON.stringify({ checks: input.rules.map(r => ({ ruleId: r.id, status: 'pass', kind: 'clause', severity: 'low', reason: `批次${input.ruleBatchIndex}`, suggestion: '', evidence: [{ blockId: 'b1', quote: '甲方验收后付款。' }], legalRefIds: [] })) }) } }] });
}
async function fixture(fn) {
  const saved = { ...config.model };
  Object.assign(config.model, { baseUrl: 'http://localhost:9999/v1', name: 'performance-fixture', key: 'fixture-only', reviewConcurrency: 3, reviewRuleBatchSize: 5 });
  try { await fn(); } finally { Object.assign(config.model, saved); }
}

test('independent review batches overlap within the configured limit and report completed work', async () => fixture(async () => {
  let active = 0, peak = 0;
  const events = [], requests = [];
  const gateway = createGateway(null, { reviewConcurrency: 2, fetchImpl: async (_, request) => {
    const input = JSON.parse(JSON.parse(request.body).messages[1].content);
    requests.push(input); peak = Math.max(peak, ++active);
    await delay(input.ruleBatchIndex % 2 ? 30 : 5); active--;
    return response(input);
  } });
  const result = await gateway.review(document, rules, [], {}, e => events.push(e));
  assert.equal(peak, 2, 'independent batches must execute concurrently');
  assert.equal(requests.length, 6);
  assert.deepEqual(result.checks.map(c => c.ruleId), rules.map(r => r.id));
  const completed = events.filter(e => e.label === 'review_batch');
  assert.deepEqual(completed.map(e => e.completedBatches), [1, 2, 3, 4, 5, 6]);
  assert.ok(completed.every(e => e.totalBatches === 6));
  assert.equal(result.coverage.complete, true);
}));

test('an irrecoverable batch failure stops new work and drains in-flight batches before returning', async () => fixture(async () => {
  let active = 0; const started = [], checkpoints = new Map();
  const gateway = createGateway(null, { reviewConcurrency: 2, fetchImpl: async (_, request) => {
    const input = JSON.parse(JSON.parse(request.body).messages[1].content); started.push(input.ruleBatchIndex); active++;
    await delay(input.ruleBatchIndex === 1 ? 5 : 35); active--;
    return input.ruleBatchIndex === 1 ? new Response('', { status: 401 }) : response(input);
  } });
  await assert.rejects(gateway.review(document, rules, [], {}, null, { save: async (key, checks) => checkpoints.set(key, checks) }), { code: 'MODEL_HTTP_401' });
  assert.equal(active, 0, 'no model work may outlive the failed review');
  assert.deepEqual(started, [1, 2]);
  assert.equal(checkpoints.size, 1, 'validated in-flight work remains resumable');
}));

test('separate gateway instances share one model request budget', async () => fixture(async () => {
  config.model.reviewConcurrency = 2;
  let active = 0, peak = 0;
  const options = { fetchImpl: async (_, request) => {
    const input = JSON.parse(JSON.parse(request.body).messages[1].content);
    peak = Math.max(peak, ++active); await delay(5); active--;
    return response(input);
  } };
  await Promise.all([createGateway(null, options).review(document, rules, [], {}), createGateway(null, options).review(document, rules, [], {})]);
  assert.equal(peak, 2);
}));

test('a review cancelled while waiting for a shared slot sends no model request', async () => fixture(async () => {
  const { withReviewSlot } = require('../pilot/review-execution');
  config.model.reviewConcurrency = 1;
  let unlock, cancelled = false, calls = 0;
  const holding = withReviewSlot(async () => { await new Promise(resolve => { unlock = resolve; }); });
  await new Promise(resolve => setImmediate(resolve));
  const store = { assertActive: async () => { if (cancelled) throw Object.assign(new Error('cancelled'), { code: 'REVIEW_CANCELLED' }); } };
  const review = createGateway(null, { fetchImpl: async (_, request) => { calls++; return response(JSON.parse(JSON.parse(request.body).messages[1].content)); } }).review(document, rules, [], {}, null, store);
  await new Promise(resolve => setImmediate(resolve)); cancelled = true; unlock();
  await holding;
  await assert.rejects(review, { code: 'REVIEW_CANCELLED' });
  assert.equal(calls, 0);
}));

test('out-of-order chunk responses preserve the original aggregation priority', async () => fixture(async () => {
  const gateway = createGateway(null, { reviewExecution: require('../pilot/review-planning').LEGACY_EXECUTION, fetchImpl: async (_, request) => {
    const input = JSON.parse(JSON.parse(request.body).messages[1].content);
    await delay(input.chunkIndex === 1 ? 20 : 1);
    return Response.json({ choices: [{ message: { content: JSON.stringify({ checks: [{ ruleId: 'r0', status: 'pass', kind: 'clause', severity: 'low', reason: `片段${input.chunkIndex}`, suggestion: '', evidence: [{ blockId: 'b1', quote: '合' }], legalRefIds: [] }] }) } }] });
  } });
  const result = await gateway.review({ blocks: [{ id: 'b1', text: '合'.repeat(17000) }] }, rules.slice(0, 1), [], {});
  assert.equal(result.checks[0].reason, '片段1');
  assert.equal(result.coverage.chunks, 2);
}));

test('progress labels count out-of-order completions, skill groups and legacy records without using stage percentages', async () => {
  const { reviewProgress } = await import('../../frontend/src/pilot/review-progress.mjs');
  const at = n => new Date(n * 1000).toISOString();
  const batch = (index, skill = 1, cached = false) => ({ stage: 'review_batch', status: 'completed', details: { chunkIndex: 1, chunkCount: 3, ruleBatchIndex: index, ruleBatchCount: 11, skillBatchIndex: skill, skillBatchCount: 2, cached } });
  const run = { stage: 'reviewing', progress: 57, created_at: at(0), steps: [{ stage: 'reviewing', created_at: at(0) }, { stage: 'review_plan', details: { totalBatches: 66, concurrency: 3 } }, batch(3), batch(2), batch(1), batch(3)] };
  const progress = reviewProgress(run, 60000);
  assert.equal(progress.completed, 3); assert.equal(progress.total, 66);
  assert.equal(progress.text, '已完成 3/66 批');
  assert.equal(progress.remainingMs, 1260000);
  run.steps.push(batch(3, 2, true));
  assert.equal(reviewProgress(run, 60000).completed, 4);
  run.steps = run.steps.filter(s => s.stage !== 'review_plan');
  assert.equal(reviewProgress(run, 60000).total, 66);
  run.steps.push({ stage: 'reviewing', created_at: at(70) });
  assert.equal(reviewProgress(run, 71000).total, 0, 'a retry starts its own batch count');
});
